import java.io.IOException;
import java.util.HashMap;
import java.util.Map;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.*;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;

public class MatrixMultiply {
    public static class M extends Mapper<LongWritable, Text, Text, Text> {
        public void map(LongWritable key, Text value, Context c)
                throws IOException, InterruptedException {
            String[] p = value.toString().trim().split(",");
            if (p.length != 4) return;
            int i = Integer.parseInt(p[1]);
            int j = Integer.parseInt(p[2]);
            if (p[0].equals("A")) {
                int cols = c.getConfiguration().getInt("b.cols", 2);
                for (int k = 0; k < cols; k++)
                    c.write(new Text(i + "," + k), new Text("A," + j + "," + p[3]));
            } else if (p[0].equals("B")) {
                int rows = c.getConfiguration().getInt("a.rows", 2);
                for (int k = 0; k < rows; k++)
                    c.write(new Text(k + "," + j), new Text("B," + i + "," + p[3]));
            }
        }
    }
    public static class R extends Reducer<Text, Text, Text, LongWritable> {
        public void reduce(Text key, Iterable<Text> values, Context c)
                throws IOException, InterruptedException {
            Map<Integer, Long> a = new HashMap<>(), b = new HashMap<>();
            for (Text v : values) {
                String[] p = v.toString().split(",");
                int k = Integer.parseInt(p[1]);
                long n = Long.parseLong(p[2]);
                (p[0].equals("A") ? a : b).merge(k, n, Long::sum);
            }
            long sum = 0;
            for (int k : a.keySet()) sum += a.get(k) * b.getOrDefault(k, 0L);
            c.write(key, new LongWritable(sum));
        }
    }
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        conf.set("fs.defaultFS", "file:///");
        conf.set("mapreduce.framework.name", "local");
        conf.setInt("a.rows", Integer.parseInt(args[2]));
        conf.setInt("b.cols", Integer.parseInt(args[3]));
        Job job = Job.getInstance(conf, "Matrix multiplication");
        job.setJarByClass(MatrixMultiply.class);
        job.setMapperClass(M.class);
        job.setReducerClass(R.class);
        job.setMapOutputKeyClass(Text.class);
        job.setMapOutputValueClass(Text.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(LongWritable.class);
        job.setNumReduceTasks(1);
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileOutputFormat.setOutputPath(job, new Path(args[1]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}

import java.io.IOException;
import java.util.*;
import org.apache.hadoop.conf.Configuration;
import org.apache.hadoop.fs.Path;
import org.apache.hadoop.io.*;
import org.apache.hadoop.mapreduce.*;
import org.apache.hadoop.mapreduce.lib.input.FileInputFormat;
import org.apache.hadoop.mapreduce.lib.input.FileSplit;
import org.apache.hadoop.mapreduce.lib.output.FileOutputFormat;

public class MovieTags {
    // Parse CSV fields, including quoted commas and escaped quotes.
    static List<String> csv(String line) {
        List<String> fields = new ArrayList<>();
        StringBuilder s = new StringBuilder();
        boolean quoted = false;
        for (int i = 0; i < line.length(); i++) {
            char ch = line.charAt(i);
            if (ch == '"') {
                if (quoted && i + 1 < line.length() && line.charAt(i+1) == '"') {
                    s.append('"'); i++;
                } else quoted = !quoted;
            } else if (ch == ',' && !quoted) {
                fields.add(s.toString()); s.setLength(0);
            } else s.append(ch);
        }
        fields.add(s.toString());
        return fields;
    }
    public static class M extends Mapper<LongWritable, Text, Text, Text> {
        public void map(LongWritable key, Text value, Context c)
                throws IOException, InterruptedException {
            List<String> p = csv(value.toString());
            String file = ((FileSplit)c.getInputSplit()).getPath().getName();
            if (file.equals("movies.csv") && p.size() >= 3 && !p.get(0).equals("movieId"))
                c.write(new Text(p.get(0)), new Text("M" + p.get(1)));
            else if (file.equals("tags.csv") && p.size() >= 4 && !p.get(0).equals("userId"))
                c.write(new Text(p.get(1)), new Text("T" + p.get(2)));
        }
    }
    public static class R extends Reducer<Text, Text, Text, Text> {
        public void reduce(Text key, Iterable<Text> values, Context c)
                throws IOException, InterruptedException {
            String title = key.toString();
            Set<String> tags = new TreeSet<>();
            for (Text v : values) {
                String s = v.toString();
                if (s.startsWith("M")) title = s.substring(1);
                else if (s.startsWith("T")) tags.add(s.substring(1));
            }
            if (!tags.isEmpty())
                c.write(new Text(key + " | " + title), new Text(String.join(", ", tags)));
        }
    }
    public static void main(String[] args) throws Exception {
        Configuration conf = new Configuration();
        conf.set("fs.defaultFS", "file:///");
        conf.set("mapreduce.framework.name", "local");
        Job job = Job.getInstance(conf, "Movie tags");
        job.setJarByClass(MovieTags.class);
        job.setMapperClass(M.class);
        job.setReducerClass(R.class);
        job.setOutputKeyClass(Text.class);
        job.setOutputValueClass(Text.class);
        job.setNumReduceTasks(1);
        FileInputFormat.addInputPath(job, new Path(args[0]));
        FileInputFormat.addInputPath(job, new Path(args[1]));
        FileOutputFormat.setOutputPath(job, new Path(args[2]));
        System.exit(job.waitForCompletion(true) ? 0 : 1);
    }
}

21. K-NEAREST NEIGHBOUR ON HANDWRITTEN DIGITS
File: knn.py
----------------------------------------
from pyspark.sql import SparkSession, Window
from pyspark.sql import functions as F
spark = SparkSession.builder.appName('KNN').getOrCreate()
df = spark.read.csv('digits.csv.gz',inferSchema=True).dropna()
data = df.select(F.array(*[F.col('_c'+str(i)).cast('double') for i in range(64)]).alias('pixels'),F.col('_c64').cast('double').alias('label'))
train,test = data.randomSplit([0.8,0.2],seed=42)
a = test.withColumn('id',F.monotonically_increasing_id()).cache()
a.count()
b = train.select(F.col('pixels').alias('train_pixels'),F.col('label').alias('neighbor_label')).withColumn('train_id',F.monotonically_increasing_id()).cache()
b.count()
pairs = a.crossJoin(F.broadcast(b)).withColumn('distance',F.expr('aggregate(zip_with(pixels, train_pixels, (x,y) -> (x-y)*(x-y)), 0D, (s,v) -> s+v)'))
w = Window.partitionBy('id').orderBy('distance','train_id')
nearest = pairs.withColumn('rank',F.row_number().over(w)).filter('rank <= 5')
votes = nearest.groupBy('id','label','neighbor_label').agg(F.count('*').alias('votes'),F.sum('distance').alias('total_distance'))
w2 = Window.partitionBy('id').orderBy(F.desc('votes'),'total_distance','neighbor_label')
pred = votes.withColumn('rank',F.row_number().over(w2)).filter('rank = 1').select('label',F.col('neighbor_label').alias('prediction')).cache()
pred.show(20)
print('Accuracy:',pred.select(F.avg((F.col('label')==F.col('prediction')).cast('double'))).first()[0])
spark.stop()
----------------------------------------
spark-submit --master 'local[*]' -i knn.scala
# Spark ML has no built-in exact KNN classifier. This uses Spark SQL cross join, squared Euclidean distance and majority voting. K=5.

val nearest = pairs.withColumn("rank",row_number().over(w)).filter("rank <= 5")
val votes = nearest.groupBy("id","label","neighbor_label").agg(count("*").alias("votes"),sum("distance").alias("total_distance"))
val w2 = Window.partitionBy("id").orderBy(desc("votes"),col("total_distance"),col("neighbor_label"))
val pred = votes.withColumn("rank",row_number().over(w2)).filter("rank = 1").select(col("label"),col("neighbor_label").alias("prediction"))
pred.show()
pred.select(avg((col("label")===col("prediction")).cast("double")).alias("accuracy")).show()
